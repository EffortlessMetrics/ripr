//! A length guard tested only at its edge.

#[derive(Debug, PartialEq)]
pub enum E {
    TooLong,
    Bad,
}

pub fn check_edge(len: usize) -> Result<usize, E> {
    if len >= 256 { return Err(E::TooLong); }
    Ok(len)
}

#[cfg(test)]
mod tests {
    use super::check_edge;

    #[test]
    fn edge_is_refused() {
        assert!(check_edge(256).is_err());
    }
}
