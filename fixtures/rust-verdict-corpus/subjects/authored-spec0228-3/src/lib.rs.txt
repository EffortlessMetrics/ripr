pub struct Gauge {
    count: u32,
    other: u32,
}

impl Gauge {
    pub fn new() -> Self {
        Gauge { count: 0, other: 0 }
    }

    pub fn step(&mut self) {
        self.count += 1;
    }

    pub fn count(&self) -> u32 {
        self.other
    }

    pub fn raw(&self) -> u32 {
        self.count
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn step_reads_the_misnamed_getter() {
        let mut c = Gauge::new();
        c.step();
        assert_eq!(c.count(), 0);
    }
}
