//! RIPR-SPEC-0225 acceptance example 12.

#[derive(Debug, Clone, PartialEq)]
pub struct Config {
    pub retries: u32,
    pub name: String,
}

pub fn build(n: u32) -> Config {
    Config {
        retries: n + 1,
        name: "x".into(),
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn builds_config() {
        let c = build(3);
        assert_eq!(c, Config { retries: c.retries, name: "x".into() });
    }
}
