//! Digit sums that refuse a non-digit character.

#[derive(Debug, PartialEq)]
pub enum E {
    TooLong,
    Bad,
}

fn digit(c: char) -> Result<u32, E> {
    c.to_digit(10).ok_or(E::Bad)
}

pub fn total(s: &str) -> Result<u32, E> {
    let mut sum = 0;
    for c in s.chars() {
        let d = digit(c)?;
        sum += d;
    }
    Ok(sum)
}

#[cfg(test)]
mod tests {
    use super::total;

    #[test]
    fn digit_is_accepted() {
        assert!(total("7").is_ok());
    }
}
