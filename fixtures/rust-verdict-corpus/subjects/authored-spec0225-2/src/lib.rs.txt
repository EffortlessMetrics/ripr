//! RIPR-SPEC-0225 acceptance example 2.

#[derive(Debug, Clone)]
pub struct Config {
    pub retries: u32,
    pub name: String,
}

impl PartialEq for Config {
    fn eq(&self, o: &Self) -> bool {
        self.name == o.name
    }
}

pub fn build(n: u32) -> Config {
    Config {
        retries: n + 1,
        name: "x".into(),
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn builds_config() {
        assert_eq!(build(3), Config { retries: 4, name: "x".into() });
    }
}
