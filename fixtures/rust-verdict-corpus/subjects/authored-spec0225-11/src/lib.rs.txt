//! RIPR-SPEC-0225 acceptance example 11.

#[derive(Debug, Clone, Copy)]
pub struct Retries(pub u32);

impl PartialEq for Retries {
    fn eq(&self, _: &Self) -> bool {
        true
    }
}

#[derive(Debug, Clone, PartialEq)]
pub struct Config {
    pub retries: Retries,
    pub name: String,
}

pub fn build(n: u32) -> Config {
    Config {
        retries: Retries(n + 1),
        name: "x".into(),
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn builds_config() {
        assert_eq!(build(3), Config { retries: Retries(4), name: "x".into() });
    }
}
