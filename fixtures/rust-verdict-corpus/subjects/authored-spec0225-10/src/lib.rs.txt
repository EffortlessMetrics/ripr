//! RIPR-SPEC-0225 acceptance example 10.

#[derive(Debug, Clone, PartialEq)]
pub struct Config {
    pub retries: u32,
    pub name: String,
}

pub fn build(n: u32) -> Config {
    Config {
        retries: n + 1,
        name: "x".into(),
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn not_nine() {
        assert_ne!(build(3), Config { retries: 9, name: "x".into() });
    }
}
