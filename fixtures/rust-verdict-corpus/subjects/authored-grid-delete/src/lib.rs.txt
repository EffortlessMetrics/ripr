//! Statements whose deletion changes state, one method per test style.

#[derive(Debug, Default)]
pub struct Stats {
    pub hits: u32,
    pub misses: u32,
    pub reads: u32,
    pub writes: u32,
    pub evictions: u32,
}

impl Stats {
    pub fn record_hit(&mut self) {
        self.hits = self.hits + 1;
    }

    pub fn record_miss(&mut self) {
        self.misses = self.misses + 1;
    }

    pub fn record_read(&mut self) {
        self.reads = self.reads + 1;
    }

    pub fn record_write(&mut self) {
        self.writes = self.writes + 1;
    }

    pub fn record_eviction(&mut self) {
        self.evictions = self.evictions + 1;
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn one_hit_counts_one() {
        let mut stats = Stats::default();
        stats.record_hit();
        assert_eq!(stats.hits, 1);
    }

    #[test]
    fn miss_table() {
        for (times, want) in [(0, 0), (1, 1), (3, 3)] {
            let mut stats = Stats::default();
            for _ in 0..times {
                stats.record_miss();
            }
            assert_eq!(stats.misses, want, "{times} misses");
        }
    }

    #[test]
    fn reads_never_decrease() {
        let mut stats = Stats::default();
        for _ in 0..10 {
            let before = stats.reads;
            stats.record_read();
            assert!(stats.reads >= before);
        }
    }

    fn check_writes(times: u32, want: u32) {
        let mut stats = Stats::default();
        for _ in 0..times {
            stats.record_write();
        }
        assert_eq!(stats.writes, want);
    }

    #[test]
    fn writes_are_counted() {
        check_writes(2, 2);
        check_writes(0, 0);
    }

    #[test]
    fn eviction_smoke() {
        let mut stats = Stats::default();
        stats.record_eviction();
    }
}
