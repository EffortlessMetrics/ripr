//! RIPR-SPEC-0225 acceptance example 8.

#[derive(Debug, Clone, PartialEq)]
pub struct Config {
    pub retries: u32,
    pub name: String,
}

pub fn build(n: u32) -> Config {
    Config {
        retries: n + 1,
        name: "x".into(),
    }
}

pub fn other() -> Config {
    Config { retries: 4, name: "x".into() }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn other_config() {
        let _c = build(3);
        assert_eq!(other(), Config { retries: 4, name: "x".into() });
    }
}
