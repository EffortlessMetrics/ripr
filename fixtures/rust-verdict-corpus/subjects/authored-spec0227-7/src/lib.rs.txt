//! A too-long error observed only by a bare unwrap_err.

#[derive(Debug, PartialEq)]
pub enum E {
    TooLong,
    Bad,
}

pub fn check_unwrap_err(len: usize) -> Result<usize, E> {
    if len >= 256 {
        return Err(E::TooLong);
    }
    Ok(len)
}

#[cfg(test)]
mod tests {
    use super::check_unwrap_err;

    #[test]
    fn unwrap_err_on_far_input() {
        let _e = check_unwrap_err(300).unwrap_err();
    }
}
