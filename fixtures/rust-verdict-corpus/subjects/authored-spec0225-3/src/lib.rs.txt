//! RIPR-SPEC-0225 acceptance example 3.

#[derive(Debug, Clone, PartialEq)]
pub struct Config {
    pub retries: u32,
    pub name: String,
}

pub fn parse(s: &str) -> Result<Config, String> {
    let n: u32 = s.parse().map_err(|_| String::from("bad"))?;
    Ok(Config {
        retries: n + 1,
        name: "x".into(),
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn parses_config() {
        assert_eq!(parse("3"), Ok(Config { retries: 4, name: "x".into() }));
    }
}
